cmake_minimum_required(VERSION 3.16)
project(p1_D0_observability)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_gtest REQUIRED)
find_package(rclcpp REQUIRED)
find_package(usv_mpc_cpp REQUIRED)
find_package(usv_research_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
function(link_regulator target)
 target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/diagnostic)
 target_link_libraries(${target} usv_mpc_cpp::mpc_core_lib OpenSSL::Crypto Threads::Threads nlohmann_json::nlohmann_json)
 ament_target_dependencies(${target} rclcpp usv_research_interfaces nav_msgs sensor_msgs std_msgs rosgraph_msgs)
endfunction()
add_executable(p1_regulator_r5_diag src/diagnostic/regulator_r5_diag.cpp)
link_regulator(p1_regulator_r5_diag)
foreach(name enqueue_branch d0_observation)
 ament_add_gtest(test_${name} tests/test_${name}.cpp TIMEOUT 30)
 link_regulator(test_${name})
 target_compile_definitions(test_${name} PRIVATE P1_REGULATOR_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/src/diagnostic/regulator_r5_diag.cpp" P1_DIAGNOSTIC_TEST)
endforeach()
