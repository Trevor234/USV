find_package(ament_cmake REQUIRED)
find_package(ament_cmake_gtest REQUIRED)

find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)

# The driver must verify every source/dependency byte before configure/build.
# These required paths never default to the live ROS workspace.
set(P1_F5_SOURCE_PACKAGE "" CACHE PATH "Immutable verified F5 usv_mpc_cpp source snapshot")
set(P1_ACADOS_DIR "" CACHE PATH "Verified immutable acados dependency root")
set(P1_CASADI_PACKAGE_DIR "" CACHE PATH "Optional verified CasADi dependency root matching F5 build capability")
set(P1_CASADI_OLD_ABI OFF CACHE BOOL "Use original F5 CasADi translation-unit ABI setting")
set(P1_D1_SOURCE_BUNDLE_SHA256 "" CACHE STRING "SHA256 of root-authored immutable derived source manifest")
string(LENGTH "${P1_D1_SOURCE_BUNDLE_SHA256}" source_sha_length)
if(NOT source_sha_length EQUAL 64 OR NOT P1_D1_SOURCE_BUNDLE_SHA256 MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "P1_D1_SOURCE_BUNDLE_SHA256 requires an exact SHA256")
endif()
if(NOT IS_ABSOLUTE "${P1_F5_SOURCE_PACKAGE}" OR NOT IS_ABSOLUTE "${P1_ACADOS_DIR}")
  message(FATAL_ERROR "Frozen F5 source and verified acados roots must be explicit absolute paths")
endif()
foreach(name mpc_solver acados_backend usv_model_dynamic utils)
  if(NOT EXISTS "${P1_F5_SOURCE_PACKAGE}/src/${name}.cpp")
    message(FATAL_ERROR "Missing frozen source ${name}.cpp")
  endif()
endforeach()
set(ACADOS_GEN_DIR "${P1_F5_SOURCE_PACKAGE}/acados_generated")
if(NOT EXISTS "${ACADOS_GEN_DIR}/acados_solver_usv_dyn_nmpc.c" OR
   NOT EXISTS "${P1_ACADOS_DIR}/lib/libacados.so")
  message(FATAL_ERROR "Frozen codegen and verified acados runtime are mandatory")
endif()
set(ACADOS_INC_DIR "${P1_ACADOS_DIR}/include")
set(ACADOS_LIB_DIR "${P1_ACADOS_DIR}/lib")
set(BLASFEO_INC_DIR "${ACADOS_INC_DIR}/blasfeo/include")
set(HPIPM_INC_DIR "${ACADOS_INC_DIR}/hpipm/include")
file(GLOB_RECURSE frozen_codegen_sources CONFIGURE_DEPENDS "${ACADOS_GEN_DIR}/*.c")
add_library(p1_frozen_codegen STATIC ${frozen_codegen_sources})
set_target_properties(p1_frozen_codegen PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
target_include_directories(p1_frozen_codegen PUBLIC ${ACADOS_GEN_DIR} ${ACADOS_INC_DIR}
  ${ACADOS_INC_DIR}/acados ${BLASFEO_INC_DIR} ${HPIPM_INC_DIR})
target_compile_options(p1_frozen_codegen PRIVATE -w)
target_link_directories(p1_frozen_codegen PUBLIC ${ACADOS_LIB_DIR})
target_link_libraries(p1_frozen_codegen PUBLIC acados hpipm blasfeo m)
add_library(p1_frozen_core STATIC "${P1_F5_SOURCE_PACKAGE}/src/usv_model_dynamic.cpp"
  "${P1_F5_SOURCE_PACKAGE}/src/utils.cpp")
target_include_directories(p1_frozen_core PUBLIC "${P1_F5_SOURCE_PACKAGE}/include")
target_link_libraries(p1_frozen_core PUBLIC Eigen3::Eigen)
set(frozen_solver_sources "${P1_F5_SOURCE_PACKAGE}/src/mpc_solver.cpp"
  "${P1_F5_SOURCE_PACKAGE}/src/acados_backend.cpp")
if(P1_CASADI_PACKAGE_DIR)
  if(NOT IS_ABSOLUTE "${P1_CASADI_PACKAGE_DIR}" OR
     NOT EXISTS "${P1_CASADI_PACKAGE_DIR}/include/casadi/casadi.hpp" OR
     NOT EXISTS "${P1_CASADI_PACKAGE_DIR}/libcasadi.so")
    message(FATAL_ERROR "Verified explicit CasADi root is incomplete")
  endif()
  list(APPEND frozen_solver_sources "${P1_F5_SOURCE_PACKAGE}/src/casadi_backend.cpp")
  if(P1_CASADI_OLD_ABI)
    set_source_files_properties("${P1_F5_SOURCE_PACKAGE}/src/mpc_solver.cpp"
      "${P1_F5_SOURCE_PACKAGE}/src/casadi_backend.cpp"
      PROPERTIES COMPILE_OPTIONS "-D_GLIBCXX_USE_CXX11_ABI=0")
  endif()
endif()
add_library(p1_frozen_solver STATIC ${frozen_solver_sources})
target_include_directories(p1_frozen_solver PUBLIC "${P1_F5_SOURCE_PACKAGE}/include"
  ${ACADOS_INC_DIR} ${BLASFEO_INC_DIR} ${HPIPM_INC_DIR} ${ACADOS_GEN_DIR})
target_link_libraries(p1_frozen_solver PUBLIC p1_frozen_core p1_frozen_codegen nlohmann_json::nlohmann_json)
if(P1_CASADI_PACKAGE_DIR)
  target_compile_definitions(p1_frozen_solver PRIVATE USV_MPC_HAS_CASADI)
  target_include_directories(p1_frozen_solver PRIVATE "${P1_CASADI_PACKAGE_DIR}/include")
  target_link_libraries(p1_frozen_solver PUBLIC "${P1_CASADI_PACKAGE_DIR}/libcasadi.so")
endif()

function(link_held_publisher target)
  target_include_directories(${target} BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_link_libraries(${target} p1_frozen_solver OpenSSL::Crypto Threads::Threads)
  ament_target_dependencies(${target} rclcpp geometry_msgs nav_msgs sensor_msgs mavros_msgs
    std_msgs rosgraph_msgs usv_research_interfaces)
  target_compile_definitions(${target} PRIVATE
    P1_D1_SOURCE_BUNDLE_SHA256="${P1_D1_SOURCE_BUNDLE_SHA256}"
    P1_D1_F5_PARENT_MANIFEST_SHA256="77737a5924f1bcd97996dee3c08e204a4a4c8fbf2f3716ff65d41fbc4bad6762")
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${ACADOS_LIB_DIR};${P1_CASADI_PACKAGE_DIR}")
endfunction()
add_executable(p1_F5_held_publisher src/mpc_node.cpp src/mpc_node_main.cpp)
link_held_publisher(p1_F5_held_publisher)
ament_add_gtest(test_publisher_startup tests/test_publisher_startup.cpp src/mpc_node.cpp TIMEOUT 45)
link_held_publisher(test_publisher_startup)
target_compile_definitions(test_publisher_startup PRIVATE P1_PUBLISHER_TEST)
set_tests_properties(test_publisher_startup PROPERTIES
  ENVIRONMENT "LD_LIBRARY_PATH=${ACADOS_LIB_DIR}:${P1_CASADI_PACKAGE_DIR}:$ENV{LD_LIBRARY_PATH};ROS_DOMAIN_ID=88;ROS_LOCALHOST_ONLY=1")
