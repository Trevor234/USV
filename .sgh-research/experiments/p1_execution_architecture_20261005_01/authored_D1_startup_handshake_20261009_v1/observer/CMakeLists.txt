cmake_minimum_required(VERSION 3.16)
project(p1_D1_shadow_ready_entry)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_gtest REQUIRED)
find_package(rclcpp REQUIRED)
find_package(usv_mpc_cpp REQUIRED)
find_package(usv_research_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
function(link_regulator target)
 target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/shadow)
 target_link_libraries(${target} usv_mpc_cpp::mpc_core_lib OpenSSL::Crypto Threads::Threads nlohmann_json::nlohmann_json)
 ament_target_dependencies(${target} rclcpp usv_research_interfaces nav_msgs sensor_msgs std_msgs rosgraph_msgs)
endfunction()
add_executable(p1_regulator_shadow_observer_ready_v1 src/shadow/regulator_shadow_observer.cpp)
link_regulator(p1_regulator_shadow_observer_ready_v1)
foreach(name shadow_entry enqueue_branch d0_observation legacy_ready)
 ament_add_gtest(test_${name} tests/test_${name}.cpp TIMEOUT 30)
 link_regulator(test_${name})
 target_compile_definitions(test_${name} PRIVATE P1_REGULATOR_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/src/shadow/regulator_shadow_observer.cpp")
endforeach()
# Test-only friends/constructors are enabled in the including test translation
# units. The production target has no test macro, install(), or workspace target.
target_compile_definitions(test_enqueue_branch PRIVATE P1_DIAGNOSTIC_TEST)
