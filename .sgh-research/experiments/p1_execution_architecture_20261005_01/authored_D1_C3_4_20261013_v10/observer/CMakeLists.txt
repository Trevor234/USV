find_package(SQLite3 REQUIRED)
function(link_regulator target)
 target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/shadow ${CMAKE_CURRENT_SOURCE_DIR}/../common)
 target_link_libraries(${target} usv_mpc_cpp::mpc_core_lib OpenSSL::Crypto Threads::Threads nlohmann_json::nlohmann_json)
 ament_target_dependencies(${target} rclcpp rmw_fastrtps_shared_cpp usv_research_interfaces nav_msgs sensor_msgs std_msgs rosgraph_msgs)
 target_compile_definitions(${target} PRIVATE P1_OWNED_SOURCE_BUNDLE_SHA256="${P1_OWNED_SOURCE_BUNDLE_SHA256}")
endfunction()
if(NOT DEFINED P1_OWNED_SOURCE_BUNDLE_SHA256)
 message(FATAL_ERROR "P1_OWNED_SOURCE_BUNDLE_SHA256 build identity required")
endif()
add_executable(p1_regulator_shadow_observer_owned_v2 src/shadow/regulator_shadow_observer.cpp)
link_regulator(p1_regulator_shadow_observer_owned_v2)
foreach(name shadow_entry enqueue_branch d0_observation atomic_observation legacy_ready c33_input_matrix c34_lease_matrix)
 ament_add_gtest(test_${name} tests/test_${name}.cpp TIMEOUT 30)
 link_regulator(test_${name})
 target_compile_definitions(test_${name} PRIVATE P1_REGULATOR_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/src/shadow/regulator_shadow_observer.cpp")
endforeach()
# Test-only friends/constructors are enabled in the including test translation
