cmake_minimum_required(VERSION 3.16)
project(p1_owned_recorder_v2)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(rosbag2_transport REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosbag2_storage REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
string(LENGTH "${P1_OWNED_SOURCE_BUNDLE_SHA256}" P1_OWNED_SOURCE_BUNDLE_SHA256_LENGTH)
if(NOT P1_OWNED_SOURCE_BUNDLE_SHA256 MATCHES "^[0-9a-f]+$" OR NOT P1_OWNED_SOURCE_BUNDLE_SHA256_LENGTH EQUAL 64)
  message(FATAL_ERROR "Require immutable -DP1_OWNED_SOURCE_BUNDLE_SHA256=<64 lowercase hex>")
endif()
add_executable(p1_owned_recorder src/owned_recorder.cpp)
target_include_directories(p1_owned_recorder PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_compile_definitions(p1_owned_recorder PRIVATE P1_OWNED_SOURCE_BUNDLE_SHA256="${P1_OWNED_SOURCE_BUNDLE_SHA256}")
target_link_libraries(p1_owned_recorder OpenSSL::Crypto nlohmann_json::nlohmann_json)
ament_target_dependencies(p1_owned_recorder rclcpp rmw_fastrtps_shared_cpp rosbag2_transport rosbag2_cpp rosbag2_storage)
