cmake_minimum_required(VERSION 3.16)
project(p1_B1_diagnostic_tools)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(usv_mpc_cpp REQUIRED)
find_package(usv_research_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(p1_regulator_r5_diag regulator_r5_diag.cpp)
target_include_directories(p1_regulator_r5_diag PRIVATE .)
target_link_libraries(p1_regulator_r5_diag usv_mpc_cpp::mpc_core_lib OpenSSL::Crypto Threads::Threads nlohmann_json::nlohmann_json)
ament_target_dependencies(p1_regulator_r5_diag rclcpp usv_research_interfaces nav_msgs sensor_msgs std_msgs rosgraph_msgs)
add_executable(p1_witness_layout witness_layout.cpp)
target_include_directories(p1_witness_layout PRIVATE .)
find_package(ament_cmake_gtest REQUIRED)
ament_add_gtest(test_p1_witness test_p1_witness.cpp)
target_include_directories(test_p1_witness PRIVATE .)
target_link_libraries(test_p1_witness Threads::Threads)
ament_add_gtest(test_p1_diag_sink test_p1_diag_sink.cpp)
target_include_directories(test_p1_diag_sink PRIVATE .)
target_link_libraries(test_p1_diag_sink Threads::Threads nlohmann_json::nlohmann_json)
# Standalone build only: there is no install into production ROS workspace.
