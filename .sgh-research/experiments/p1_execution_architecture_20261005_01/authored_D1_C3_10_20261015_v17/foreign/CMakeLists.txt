cmake_minimum_required(VERSION 3.16)
project(p1_c310_foreign_applied LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(usv_research_interfaces REQUIRED)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
set(P1_OWNED_SOURCE_BUNDLE_SHA256 "" CACHE STRING "Exact independent producer source bundle")
string(LENGTH "${P1_OWNED_SOURCE_BUNDLE_SHA256}" pin_length)
if(NOT pin_length EQUAL 64)
  message(FATAL_ERROR "Exact source bundle SHA required")
endif()
add_executable(p1_c310_foreign_applied foreign_applied.cpp)
target_include_directories(p1_c310_foreign_applied PRIVATE common .)
target_compile_definitions(p1_c310_foreign_applied PRIVATE P1_OWNED_SOURCE_BUNDLE_SHA256="${P1_OWNED_SOURCE_BUNDLE_SHA256}")
target_link_libraries(p1_c310_foreign_applied OpenSSL::Crypto nlohmann_json::nlohmann_json)
ament_target_dependencies(p1_c310_foreign_applied rclcpp usv_research_interfaces rmw_fastrtps_shared_cpp)
find_package(ament_cmake_gtest REQUIRED)
ament_add_gtest(test_c310_foreign_protocol tests/test_foreign_protocol.cpp TIMEOUT 30)
target_include_directories(test_c310_foreign_protocol PRIVATE common .)
target_link_libraries(test_c310_foreign_protocol OpenSSL::Crypto nlohmann_json::nlohmann_json)
ament_target_dependencies(test_c310_foreign_protocol rclcpp usv_research_interfaces rmw_fastrtps_shared_cpp)
