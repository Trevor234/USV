cmake_minimum_required(VERSION 3.16)
project(p1_enqueue_fix_tools)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_gtest REQUIRED)
find_package(rclcpp REQUIRED)
find_package(usv_mpc_cpp REQUIRED)
find_package(usv_research_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
function(link_regulator target dir)
 target_include_directories(${target} PRIVATE ${dir})
 target_link_libraries(${target} usv_mpc_cpp::mpc_core_lib OpenSSL::Crypto Threads::Threads nlohmann_json::nlohmann_json)
 ament_target_dependencies(${target} rclcpp usv_research_interfaces nav_msgs sensor_msgs std_msgs rosgraph_msgs)
endfunction()
add_executable(p1_regulator src/ordinary/regulator.cpp)
link_regulator(p1_regulator ${CMAKE_CURRENT_SOURCE_DIR}/src/ordinary)
add_executable(p1_regulator_r5_diag src/diagnostic/regulator_r5_diag.cpp)
link_regulator(p1_regulator_r5_diag ${CMAKE_CURRENT_SOURCE_DIR}/src/diagnostic)
foreach(flavor ordinary diagnostic)
 if(flavor STREQUAL "ordinary")
  set(source_name regulator.cpp)
 else()
  set(source_name regulator_r5_diag.cpp)
 endif()
 foreach(version src negative)
  set(target test_enqueue_${version}_${flavor})
  ament_add_gtest(${target} tests/test_enqueue_branch.cpp TIMEOUT 30)
  set(dir ${CMAKE_CURRENT_SOURCE_DIR}/${version}/${flavor})
  link_regulator(${target} ${dir})
  target_compile_definitions(${target} PRIVATE P1_REGULATOR_SOURCE="${dir}/${source_name}")
  if(flavor STREQUAL "diagnostic")
   target_compile_definitions(${target} PRIVATE P1_DIAGNOSTIC_TEST)
  endif()
 endforeach()
endforeach()
