cmake_minimum_required(VERSION 3.16)
project(model_setter_review LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 99)
get_filename_component(REVIEW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(IMPL "${REVIEW_ROOT}/implementation")
set(ACADOS /home/trevor/acados)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
file(GLOB_RECURSE GENERATED "${IMPL}/baseline/acados_generated/*.c")
add_library(generated_current STATIC ${GENERATED})
target_include_directories(generated_current PUBLIC "${IMPL}/baseline/acados_generated" "${ACADOS}/include" "${ACADOS}/include/acados" "${ACADOS}/include/blasfeo/include" "${ACADOS}/include/hpipm/include")
target_compile_options(generated_current PRIVATE -w)
target_link_directories(generated_current PUBLIC "${ACADOS}/lib")
target_link_libraries(generated_current PUBLIC acados hpipm blasfeo m)
set(WRAPS usv_dyn_nmpc_acados_solve usv_dyn_nmpc_acados_set_p_global_and_precompute_dependencies usv_dyn_nmpc_acados_create_capsule usv_dyn_nmpc_acados_free usv_dyn_nmpc_acados_free_capsule)
foreach(VARIANT baseline packing_only checked_init)
  add_library(backend_${VARIANT} STATIC "${IMPL}/${VARIANT}/src/acados_backend.cpp" "${IMPL}/${VARIANT}/src/utils.cpp")
  target_include_directories(backend_${VARIANT} PUBLIC "${IMPL}/${VARIANT}/include")
  target_link_libraries(backend_${VARIANT} PUBLIC generated_current Eigen3::Eigen Threads::Threads)
  add_executable(behavior_${VARIANT} "${IMPL}/tests/test_backend_behavior.cpp")
  target_link_libraries(behavior_${VARIANT} PRIVATE backend_${VARIANT})
  foreach(SYMBOL ${WRAPS})
    target_link_options(behavior_${VARIANT} PRIVATE "-Wl,--wrap=${SYMBOL}")
  endforeach()
endforeach()

target_link_options(experimental_f6 PRIVATE "-Wl,--wrap=usv_dyn_nmpc_acados_solve")
