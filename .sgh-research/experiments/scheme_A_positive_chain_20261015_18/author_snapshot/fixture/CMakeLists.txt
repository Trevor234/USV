cmake_minimum_required(VERSION 3.16)
project(p1_owned_synthetic_fixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
set(P1_OWNED_SOURCE_BUNDLE_SHA256 "" CACHE STRING "Pinned owned source bundle")
string(LENGTH "${P1_OWNED_SOURCE_BUNDLE_SHA256}" pin_length)
if(NOT pin_length EQUAL 64)
  message(FATAL_ERROR "Exact source bundle SHA required")
endif()
add_executable(p1_synthetic_source synthetic_source.cpp)
target_include_directories(p1_synthetic_source PRIVATE ../common)
target_compile_definitions(p1_synthetic_source PRIVATE P1_OWNED_SOURCE_BUNDLE_SHA256="${P1_OWNED_SOURCE_BUNDLE_SHA256}")
target_link_libraries(p1_synthetic_source OpenSSL::Crypto nlohmann_json::nlohmann_json)
ament_target_dependencies(p1_synthetic_source rclcpp nav_msgs sensor_msgs std_msgs rosgraph_msgs rmw_fastrtps_shared_cpp)

# Pure message/CDR finite checks; this test starts no ROS node or DDS participant.
find_package(ament_cmake_gtest REQUIRED)
ament_add_gtest(test_zero_reference test_zero_reference.cpp TIMEOUT 30)
ament_target_dependencies(test_zero_reference rclcpp nav_msgs std_msgs)
