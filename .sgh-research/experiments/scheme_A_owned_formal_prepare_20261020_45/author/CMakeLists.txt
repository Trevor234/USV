cmake_minimum_required(VERSION 3.16)
project(p1_owned_formal_prepare)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(usv_research_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
if(NOT DEFINED P1_OWNED_SOURCE_BUNDLE_SHA256 OR NOT DEFINED P1_DRIVER_TOOL_SHA256)
 message(FATAL_ERROR "source bundle and Driver SHA required")
endif()
foreach(role fixture preparer)
 add_executable(p1_formal_${role}_owned_v1 native/formal_${role}.cpp)
 target_include_directories(p1_formal_${role}_owned_v1 PRIVATE common)
 target_link_libraries(p1_formal_${role}_owned_v1 OpenSSL::Crypto nlohmann_json::nlohmann_json)
 ament_target_dependencies(p1_formal_${role}_owned_v1 rclcpp rmw_fastrtps_shared_cpp usv_research_interfaces nav_msgs sensor_msgs std_msgs rosgraph_msgs)
 target_compile_definitions(p1_formal_${role}_owned_v1 PRIVATE P1_OWNED_SOURCE_BUNDLE_SHA256="${P1_OWNED_SOURCE_BUNDLE_SHA256}" P1_DRIVER_TOOL_SHA256="${P1_DRIVER_TOOL_SHA256}")
endforeach()
add_executable(test_formal_input_bounds native/formal_input_bounds_test.cpp)
target_include_directories(test_formal_input_bounds PRIVATE common)
target_link_libraries(test_formal_input_bounds OpenSSL::Crypto nlohmann_json::nlohmann_json)
ament_target_dependencies(test_formal_input_bounds rclcpp rmw_fastrtps_shared_cpp usv_research_interfaces nav_msgs sensor_msgs std_msgs rosgraph_msgs)
target_compile_definitions(test_formal_input_bounds PRIVATE P1_DRIVER_TOOL_SHA256="${P1_DRIVER_TOOL_SHA256}")
