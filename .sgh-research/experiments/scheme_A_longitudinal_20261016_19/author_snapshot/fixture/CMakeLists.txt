cmake_minimum_required(VERSION 3.16)
project(p1_controlled_phase_fixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(usv_research_interfaces REQUIRED)
find_package(ament_cmake_gtest REQUIRED)
set(P1_OWNED_SOURCE_BUNDLE_SHA256 "" CACHE STRING "Pinned owned source bundle")
string(LENGTH "${P1_OWNED_SOURCE_BUNDLE_SHA256}" pin_length)
if(NOT pin_length EQUAL 64)
  message(FATAL_ERROR "Exact source bundle SHA required")
endif()
add_executable(p1_controlled_phase_fixture synthetic_source.cpp ../owner_common/usv_model_dynamic.cpp)
target_include_directories(p1_controlled_phase_fixture PRIVATE ../owner_common)
target_compile_definitions(p1_controlled_phase_fixture PRIVATE P1_OWNED_SOURCE_BUNDLE_SHA256="${P1_OWNED_SOURCE_BUNDLE_SHA256}")
target_link_libraries(p1_controlled_phase_fixture OpenSSL::Crypto nlohmann_json::nlohmann_json Eigen3::Eigen)
ament_target_dependencies(p1_controlled_phase_fixture rclcpp nav_msgs sensor_msgs std_msgs rosgraph_msgs rmw_fastrtps_shared_cpp usv_research_interfaces)

ament_add_gtest(test_controlled_dynamic_fixture test_controlled_dynamic_fixture.cpp ../owner_common/usv_model_dynamic.cpp TIMEOUT 30)
target_include_directories(test_controlled_dynamic_fixture PRIVATE ../owner_common)
target_link_libraries(test_controlled_dynamic_fixture OpenSSL::Crypto nlohmann_json::nlohmann_json Eigen3::Eigen)
ament_target_dependencies(test_controlled_dynamic_fixture rclcpp nav_msgs sensor_msgs std_msgs rosgraph_msgs rmw_fastrtps_shared_cpp usv_research_interfaces)
