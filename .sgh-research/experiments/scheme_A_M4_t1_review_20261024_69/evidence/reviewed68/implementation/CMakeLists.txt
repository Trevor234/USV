function(link_t1 target)
  target_include_directories(${target} PRIVATE ${T1_AUTHOR_DIR}/observer/src/shadow ${T1_AUTHOR_DIR}/common ${T1_AUTHOR_DIR}/worker ${T1_AUTHOR_DIR}/fixture)
  target_link_libraries(${target} usv_mpc_cpp::mpc_core_lib OpenSSL::Crypto Threads::Threads nlohmann_json::nlohmann_json)
  ament_target_dependencies(${target} rclcpp rmw_fastrtps_shared_cpp usv_research_interfaces nav_msgs sensor_msgs std_msgs rosgraph_msgs)
  target_compile_definitions(${target} PRIVATE
    P1_OWNED_SOURCE_BUNDLE_SHA256="${P1_OWNED_SOURCE_BUNDLE_SHA256}"
    P1_DRIVER_TOOL_SHA256="${P1_DRIVER_TOOL_SHA256}"
    P1_FROZEN_CONTRACT="${P1_FROZEN_CONTRACT}")
endfunction()
