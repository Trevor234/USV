cmake_minimum_required(VERSION 3.16)
project(factory_routing_review LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(IMPL "${ROOT}/implementation")
set(ACADOS /home/trevor/acados)
set(CASADI /home/trevor/.local/lib/python3.12/site-packages/casadi)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
add_library(generated_current STATIC IMPORTED)
set_target_properties(generated_current PROPERTIES IMPORTED_LOCATION "${IMPL}/libgenerated_current.a")
set(INCS "${IMPL}/baseline/acados_generated" "${ACADOS}/include" "${ACADOS}/include/acados" "${ACADOS}/include/blasfeo/include" "${ACADOS}/include/hpipm/include")
foreach(variant baseline checked_init)
  add_library(backend_${variant} STATIC "${IMPL}/${variant}/src/acados_backend.cpp" "${IMPL}/${variant}/src/utils.cpp")
  target_include_directories(backend_${variant} PUBLIC "${IMPL}/${variant}/include" ${INCS})
  target_link_directories(backend_${variant} PUBLIC "${ACADOS}/lib")
  # The CasADi wheel also ships libhpipm/libblasfeo. Select the registered
  # acados dependencies by absolute path rather than ambiguous -l resolution.
  target_link_libraries(backend_${variant} PUBLIC generated_current "${ACADOS}/lib/libacados.so" "${ACADOS}/lib/libhpipm.so" "${ACADOS}/lib/libblasfeo.so" m Eigen3::Eigen Threads::Threads)
endforeach()
add_library(casadi_real STATIC "${IMPL}/current/src/casadi_backend.cpp")
target_include_directories(casadi_real PRIVATE "${IMPL}/checked_init/include" "${IMPL}/current/include" "${CASADI}/include")
target_link_libraries(casadi_real PRIVATE Eigen3::Eigen)
target_compile_options(casadi_real PRIVATE -D_GLIBCXX_USE_CXX11_ABI=0)
set(WRAPS usv_dyn_nmpc_acados_solve usv_dyn_nmpc_acados_set_p_global_and_precompute_dependencies usv_dyn_nmpc_acados_create_capsule usv_dyn_nmpc_acados_free usv_dyn_nmpc_acados_free_capsule)
foreach(variant baseline checked_init admission_guard)
  foreach(mode off on)
    if(variant STREQUAL admission_guard)
      set(FACTORY "${IMPL}/admission_guard/src/mpc_solver.cpp")
      set(BACKEND checked_init)
    else()
      set(FACTORY "${IMPL}/current/src/mpc_solver.cpp")
      set(BACKEND ${variant})
    endif()
    # The guard is a conditional target: prepare it only after reproduction.
    if(NOT EXISTS "${FACTORY}")
      continue()
    endif()
    set(target factory_${variant}_${mode})
    add_library(route_${variant}_${mode} OBJECT "${FACTORY}")
    target_include_directories(route_${variant}_${mode} PRIVATE "${IMPL}/checked_init/include" "${IMPL}/current/include" ${INCS})
    target_link_libraries(route_${variant}_${mode} PRIVATE Eigen3::Eigen)
    add_executable(${target} "${IMPL}/tests/test_factory.cpp" $<TARGET_OBJECTS:route_${variant}_${mode}>)
    target_include_directories(${target} PRIVATE "${IMPL}/checked_init/include" "${IMPL}/current/include")
    target_link_libraries(${target} PRIVATE backend_${BACKEND})
    if(mode STREQUAL on)
      target_compile_definitions(${target} PRIVATE USV_MPC_HAS_CASADI)
      target_compile_definitions(route_${variant}_${mode} PRIVATE USV_MPC_HAS_CASADI)
      target_compile_options(route_${variant}_${mode} PRIVATE -D_GLIBCXX_USE_CXX11_ABI=0)
      target_include_directories(route_${variant}_${mode} PRIVATE "${CASADI}/include")
      target_include_directories(${target} PRIVATE "${CASADI}/include")
      target_link_directories(${target} PRIVATE "${CASADI}")
      target_link_libraries(${target} PRIVATE casadi_real casadi)
    endif()
    set_target_properties(${target} PROPERTIES BUILD_RPATH "${ACADOS}/lib;${CASADI}")
    foreach(symbol ${WRAPS})
      target_link_options(${target} PRIVATE "-Wl,--wrap=${symbol}")
    endforeach()
  endforeach()
endforeach()
